UNIT neorv32_cpu_icc
  INPUTS: clk_i, csr_addr_i, csr_re_i, csr_wdata_i, csr_we_i, icc_rx_i, rstn_i
  OUTPUTS: csr_rdata_o, icc_tx_o
  BLOCKS
    [inst] tx_queue_inst (neorv32_fifo, another IP): tx_fifo_free
    [logic] tx_fifo_we: tx_fifo_we
    [process] csr_read: -
  ARROWS (==> a value moves; --> only decides; [the elements the arrow carries])
    clk_i ==> tx_queue_inst   [clk_i]
    csr_addr_i --> tx_fifo_we   [csr_addr_i]
    csr_addr_i --> icc_tx_o   [csr_addr_i]
    csr_addr_i --> csr_read   [csr_addr_i]
    csr_re_i --> icc_tx_o   [csr_re_i]
    csr_wdata_i ==> tx_queue_inst   [csr_wdata_i]
    csr_we_i --> tx_fifo_we   [csr_we_i]
    icc_rx_i ==> tx_queue_inst   [icc_rx_i.ack]
    icc_rx_i ==> csr_read   [icc_rx_i.{dat,rdy}]
    rstn_i ==> tx_queue_inst   [rstn_i]
    tx_queue_inst ==> icc_tx_o   [avail_o, rdata_o]
    tx_queue_inst ==> csr_read   [tx_fifo_free]
    tx_fifo_we ==> tx_queue_inst   [tx_fifo_we]
    csr_read ==> csr_rdata_o
