UNIT neorv32_cpu_cp_cond
  INPUTS: clk_i, ctrl_i, rs1_i, rs2_i, rstn_i
  OUTPUTS: res_o, valid_o
  BLOCKS
    [logic] condition: condition
    [logic] rs2_zero: rs2_zero
    [logic] valid_cmd: valid_cmd
    [process] cond_out: -; clocked by clk_i; reset by rstn_i
  ARROWS (==> a value moves; --> only decides; [the elements the arrow carries])
    ctrl_i ==> condition   [ctrl_i.ir_funct3]
    ctrl_i --> valid_cmd   [ctrl_i.{alu_cp_alu,ir_funct12,ir_funct3,ir_opcode}]
    rs1_i ==> cond_out   [rs1_i]
    rs2_i --> rs2_zero   [rs2_i]
    condition --> cond_out   [condition]
    rs2_zero ==> condition   [rs2_zero]
    valid_cmd ==> valid_o   [valid_cmd]
    valid_cmd --> cond_out   [valid_cmd]
    cond_out ==> res_o
