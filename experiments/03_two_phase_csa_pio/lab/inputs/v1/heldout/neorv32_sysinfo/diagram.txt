UNIT neorv32_sysinfo
  INPUTS: bus_req_i, clk_i, rstn_i
  OUTPUTS: bus_rsp_o
  BLOCKS
    [logic] sysinfo: sysinfo
    [process] access_buffer: buf_ack, buf_adr; clocked by clk_i; reset by rstn_i
    [process] sysinfo_clk: sysinfo; clocked by clk_i; reset by rstn_i
  ARROWS (==> a value moves; --> only decides; [the elements the arrow carries])
    bus_req_i ==> access_buffer   [bus_req_i.{addr,stb}]
    bus_req_i ==> sysinfo_clk   [bus_req_i.{addr,data,rw,stb}]
    sysinfo ==> bus_rsp_o   [sysinfo]
    access_buffer ==> bus_rsp_o   [buf_ack, buf_adr]
    sysinfo_clk ==> bus_rsp_o   [sysinfo]
