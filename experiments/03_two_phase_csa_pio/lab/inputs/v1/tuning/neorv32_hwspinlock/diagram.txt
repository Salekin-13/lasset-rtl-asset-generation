UNIT neorv32_hwspinlock
  INPUTS: bus_req_i, clk_i, rstn_i
  OUTPUTS: bus_rsp_o
  BLOCKS
    [generate] spinlock_gen: sel
    [process] bus_response: -; clocked by clk_i; reset by rstn_i
    [process] spinlock: lock_q; clocked by clk_i; reset by rstn_i
  ARROWS (==> a value moves; --> only decides; [the elements the arrow carries])
    spinlock_gen ==> bus_response   [sel]
    spinlock_gen --> spinlock   [sel]
    bus_req_i --> spinlock_gen   [bus_req_i.addr]
    bus_req_i ==> bus_response   [bus_req_i.{addr,rw,stb}]
    bus_req_i ==> spinlock   [bus_req_i.{addr,rw,stb}]
    bus_response ==> bus_rsp_o
    spinlock ==> bus_response   [lock_q]
