   74 |     quotient  : std_ulogic_vector(XLEN-1 downto 0);
   75 |     sub       : std_ulogic_vector(XLEN   downto 0);
   76 |     res_u     : std_ulogic_vector(XLEN-1 downto 0);
   77 |     res       : std_ulogic_vector(XLEN-1 downto 0);
   78 |   end record;
   79 |   signal div : div_t;
   82 |   type mul_t is record
   83 |     start  : std_ulogic;
   84 |     prod   : std_ulogic_vector((2*XLEN)-1 downto 0);
   85 |     add    : std_ulogic_vector(XLEN downto 0);
   86 |     p_sext : std_ulogic;
   87 |     dsp_x  : signed(XLEN downto 0);
   88 |     dsp_y  : signed(XLEN downto 0);
   89 |     dsp_z  : signed(2*XLEN+1 downto 0);
   90 |   end record;
   91 |   signal mul : mul_t;
   93 | begin
   97 |   valid_cmd <= '1' when (ctrl_i.alu_cp_alu = '1') and (ctrl_i.ir_opcode(5) = '1') and
   98 |                         (ctrl_i.ir_funct12(11 downto 5) = "0000001") and
   99 |                         ((ctrl_i.ir_funct3(2) = '0') or ((ctrl_i.ir_funct3(2) = '1') and DIVISION_EN)) else '0';

  105 |   begin

  108 |       ctrl.cnt    <= (others => '0');
  109 |       ctrl.out_en <= '0';

  142 |   end process control;

  150 |   ctrl.rs2_is_signed <= '1' when (ctrl_i.ir_funct3 = op_mulh_c) or

  154 |   mul.start <= '1' when (valid_cmd = '1') and (ctrl_i.ir_funct3(2) = '0') else '0';

  160 |   multiplier_core_parallel:

  164 |     multiplier_core_in_reg: process(rstn_i, clk_i)
  165 |     begin
  166 |       if (rstn_i = '0') then
  167 |         mul.dsp_x <= (others => '0');
  168 |         mul.dsp_y <= (others => '0');
  169 |       elsif rising_edge(clk_i) then
  170 |         if (mul.start = '1') then
  171 |           mul.dsp_x <= signed((rs1_i(rs1_i'left) and ctrl.rs1_is_signed) & rs1_i);
  172 |           mul.dsp_y <= signed((rs2_i(rs2_i'left) and ctrl.rs2_is_signed) & rs2_i);
  173 |         end if;
  174 |       end if;
  175 |     end process multiplier_core_in_reg;
  178 |     mul.dsp_z <= mul.dsp_x * mul.dsp_y;
  181 |     multiplier_core_out_reg: process(clk_i)
  182 |     begin
  183 |       if rising_edge(clk_i) then
  184 |         mul.prod <= std_ulogic_vector(mul.dsp_z(63 downto 0));
  185 |       end if;
  186 |     end process multiplier_core_out_reg;
  188 |   end generate;
  191 |   multiplier_core_parallel_none:
  192 |   if not FAST_MUL_EN generate
  193 |     mul.dsp_x <= (others => '0');
  194 |     mul.dsp_y <= (others => '0');
  195 |     mul.dsp_z <= (others => '0');
  196 |   end generate;
  201 |   multiplier_core_serial:
  202 |   if not FAST_MUL_EN generate
  205 |     multiplier_core: process(rstn_i, clk_i)
  206 |     begin
  207 |       if (rstn_i = '0') then

  209 |       elsif rising_edge(clk_i) then

  211 |           mul.prod(63 downto 32) <= (others => '0');
  212 |           mul.prod(31 downto 0)  <= rs1_i;
  213 |         elsif (ctrl.state /= S_IDLE) then

  215 |           mul.prod(30 downto 0)  <= mul.prod(31 downto 1);
  216 |         end if;
  217 |       end if;
  218 |     end process multiplier_core;
  221 |     mul_update: process(mul, ctrl, rs2_i)
  222 |     begin

  224 |         if (ctrl.state = S_DONE) and (ctrl.rs1_is_signed = '1') then
  225 |           mul.add <= std_ulogic_vector(unsigned(mul.p_sext & mul.prod(63 downto 32)) - unsigned((rs2_i(rs2_i'left) and ctrl.rs2_is_signed) & rs2_i));
  226 |         else
  227 |           mul.add <= std_ulogic_vector(unsigned(mul.p_sext & mul.prod(63 downto 32)) + unsigned((rs2_i(rs2_i'left) and ctrl.rs2_is_signed) & rs2_i));

  229 |       else
  230 |         mul.add <= mul.p_sext & mul.prod(63 downto 32);
  231 |       end if;
  232 |     end process mul_update;
  235 |     mul.p_sext <= mul.prod(mul.prod'left) and ctrl.rs2_is_signed;
  237 |   end generate;
  240 |   multiplier_core_serial_none:
  241 |   if FAST_MUL_EN generate
  242 |     mul.add    <= (others => '0');
  243 |     mul.p_sext <= '0';
  244 |   end generate;
  249 |   divider_core_serial:
  250 |   if DIVISION_EN generate
  253 |     divider_core: process(rstn_i, clk_i)
  254 |     begin
  255 |       if (rstn_i = '0') then
  256 |         div.quotient  <= (others => '0');

  258 |         div.sign_mod  <= '0';
  259 |         div.remainder <= (others => '0');

  280 |             div.sign_mod <= '0';

  291 |         end if;

  302 |   end generate;
  305 |   divider_core_serial_none:
  306 |   if not DIVISION_EN generate
  307 |     div.quotient  <= (others => '0');
  308 |     div.sign_mod  <= '0';
  309 |     div.rs2_abs   <= (others => '0');
  310 |     div.remainder <= (others => '0');
  311 |     div.sub       <= (others => '0');
  312 |     div.res_u     <= (others => '0');
  313 |     div.res       <= (others => '0');
  314 |   end generate;
