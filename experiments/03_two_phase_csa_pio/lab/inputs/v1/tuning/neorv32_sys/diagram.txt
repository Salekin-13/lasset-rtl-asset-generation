UNIT neorv32_sys_reset
  INPUTS: clk_i, rstn_dbg_i, rstn_ext_i, rstn_wdt_i
  OUTPUTS: rstn_ext_o, rstn_sys_o, xrstn_ocd_o, xrstn_wdt_o
  BLOCKS
    [process] sequencer: sreg_ext, sreg_sys; clocked by clk_i; reset by rstn_ext_i
    [process] synchronizer: -; clocked by clk_i; reset by rstn_ext_i
  ARROWS (==> a value moves; --> only decides; [the elements the arrow carries])
    rstn_dbg_i --> sequencer   [rstn_dbg_i]
    rstn_dbg_i ==> synchronizer   [rstn_dbg_i]
    rstn_wdt_i --> sequencer   [rstn_wdt_i]
    rstn_wdt_i ==> synchronizer   [rstn_wdt_i]
    sequencer ==> rstn_ext_o
    sequencer ==> rstn_sys_o
    synchronizer ==> xrstn_ocd_o
    synchronizer ==> xrstn_wdt_o

UNIT neorv32_sys_clock
  INPUTS: clk_i, enable_i, rstn_i
  OUTPUTS: clk_en_o
  BLOCKS
    [process] ticker: cnt, cnt2, en; clocked by clk_i; reset by rstn_i
  ARROWS (==> a value moves; --> only decides; [the elements the arrow carries])
    enable_i ==> ticker   [enable_i]
    ticker ==> clk_en_o   [cnt, cnt2]
